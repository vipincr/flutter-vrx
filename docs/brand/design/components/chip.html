Resolved rules

The layers, base first; each later layer wins
basecontainer: fill = colour.secondary_container, radius = radius_mm.sm, border = border_px.regular, border_colour = colour.outline, pad_x = spacing_mm.md, pad_y = spacing_mm.xs, gap = spacing_mm.xs, hit = control_mm.sm, motion = micro.select; label: type = type.label, colour = colour.on_secondary_container; icon: size = icon_mm.xs, colour = colour.on_secondary_container
variant assistno change
variant filterno change
size smcontainer: height = control_mm.sm
size mdcontainer: height = control_mm.md; label: type = type.label_lg
state restno change
state hovercontainer: layer = colour.primary, layer_opacity = opacity_permille.hover
state pressedcontainer: layer = colour.primary, layer_opacity = class.pressed_layer_permille
state selectedcontainer: fill = colour.primary, border_colour = colour.primary; label: colour = colour.on_primary; icon: colour = colour.on_primary
state focusfocus_ring: colour = colour.focus, stroke = border_px.regular, offset = spacing_mm.xs
state disabledcontainer: opacity = opacity_permille.disabled
