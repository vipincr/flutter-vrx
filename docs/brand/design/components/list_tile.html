Resolved rules

The layers, base first; each later layer wins
basecontainer: pad_x = spacing_mm.edge_inset, pad_y = spacing_mm.sm, gap = spacing_mm.md, hit = control_mm.md, motion = micro.press_row; title: type = type.body, colour = colour.text; subtitle: type = type.subtext, colour = colour.text_secondary; leading: size = icon_mm.md, colour = colour.text_secondary; trailing: size = icon_mm.sm, colour = colour.text_secondary
variant plainno change
size one_linecontainer: height = control_mm.md
size two_linecontainer: height = control_mm.lg
state restno change
state hovercontainer: layer = colour.primary, layer_opacity = opacity_permille.hover
state pressedcontainer: layer = colour.primary, layer_opacity = class.pressed_layer_permille
state selectedcontainer: fill = colour.primary; title: colour = colour.on_primary; subtitle: colour = colour.on_primary; leading: colour = colour.on_primary; trailing: colour = colour.on_primary
state focusfocus_ring: colour = colour.focus, stroke = border_px.regular, offset = spacing_mm.xs
state disabledcontainer: opacity = opacity_permille.disabled
